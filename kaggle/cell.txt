def banner(text):
    print()
    print("=" * 70)
    print(text)
    print("=" * 70)


def step_ok(name, extra=""):
    suffix = f" ({extra})" if extra else ""
    print(f"[OK  ] {name}{suffix}")


def step_fail(name, detail):
    print(f"[FAIL] {name}: {detail}")


def show_state(label):
    if not STATE_DB.exists():
        return f"no state yet: {label}"
    try:
        con = sqlite3.connect(STATE_DB)
        rows = con.execute("select status, count(*) from videos group by status").fetchall()
        con.close()
    except sqlite3.Error as exc:
        print("could not read state db:", exc)
        return f"unreadable state: {label}"
    if not rows:
        return f"empty state: {label}"
    return f"{label}: " + ", ".join(f"{status}={count}" for status, count in rows)

    step_ok("project already extracted", str(PROJECT))
else:
    zips = [
        p
        for p in glob.glob("/kaggle/input/**/*.zip", recursive=True)
        + glob.glob("/kaggle/input/**/*.ZIP", recursive=True)
        if "plainbagel" in p.lower()
    ]
    zips.sort(key=lambda p: ("plainbagel-archiver" not in p.lower(), p))
    if zips:
        zip_path = zips[0]
        print(f"using zip: {zip_path}")

        shutil.rmtree(PROJECT, ignore_errors=True)

                "again to https://www.kaggle.com/datasets/nekomeowfm/plainbagel-archiver",

def find_cookie_files(base: str) -> list[str]:
    """Find Netscape cookie .txt files under base, whatever their filename is."""
    hits: list[str] = []
    for root, _dirs, names in os.walk(base):
        for name in names:
            if "cookie" not in name.lower() or not name.lower().endswith(".txt"):
                continue
            path = os.path.join(root, name)
            try:
                with open(path, "rb") as handle:
                    if not handle.read(64).lstrip().startswith(b"#"):
                        continue
            except OSError:
                continue
            hits.append(path)
    return sorted(hits)


cookie_files = find_cookie_files("/kaggle/input")
cookie_args: list[str] = []
if cookie_files:
    print("cookie file(s) present:", ", ".join(os.path.basename(p) for p in cookie_files))
    cookie_args = ["--cookies", cookie_files[0]]
    step_ok("using YouTube cookies file automatically", cookie_files[0])
else:
    print("no Netscape .txt cookie file under /kaggle/input; anonymous sessions will be used")
    loose = sorted(
        os.path.join(root, name)
        for root, _dirs, fs in os.walk("/kaggle/input")
        for name in fs
        if "cookie" in name.lower()
    )
    if loose:
        print("  files that look like cookies but are not Netscape .txt:",
              ", ".join(os.path.basename(p) for p in loose))

desired_clients = os.environ.get("PB_CLIENTS", "").strip()
if not desired_clients and cookie_args:
    # with valid cookies the widest compatible client set is safest for media downloads
    desired_clients = "youtube:player_client=default,web,web_safari,tv,tv_simply"
if desired_clients:
    config_path = PROJECT / "config.toml"
    lines = config_path.read_text(encoding="utf-8").splitlines()
    patched = False
    for index, line in enumerate(lines):
        if line.strip().startswith("ytdlp_extractor_args"):
            lines[index] = f'ytdlp_extractor_args = "{desired_clients}"'
            patched = True
            break
    if patched:
        config_path.write_text("\n".join(lines) + "\n", encoding="utf-8")
        step_ok("yt-dlp clients:", desired_clients)
    else:
        step_fail(
            "yt-dlp clients override",
            "config.toml in the delivered zip has no ytdlp_extractor_args key; "
            "upload the newest plainbagel-archiver.zip as a new dataset version",
        )

session_minutes = os.environ.get("PB_SESSION_MINUTES", "").strip() or "660"


def archiver(*args, capture=False):
    command = [
        sys.executable,
        "-m",
        "plainbagel_archiver",
        "--config",
        str(PROJECT / "config.toml"),
        "--work-dir",
        str(WORK),
        *args,
    ]
    print("$", " ".join(command))
    if capture:
        return subprocess.run(command, capture_output=True, text=True, check=False)
    return subprocess.run(command, check=False)


doctor = archiver("doctor")
if doctor.returncode != 0:
    step_fail("doctor", f"exit {doctor.returncode}; fix the problem above before continuing")
    raise SystemExit(1)

        except sqlite3.Error:
